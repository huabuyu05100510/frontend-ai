## 一、实验框架设计

### 1.1 实验目标

**核心问题**：voice-kit 的技术方案是否优于传统方案？

**对比维度**：
1. **性能**：延迟、吞吐、资源占用
2. **质量**：丢帧率、卡顿率、音频质量
3. **稳定性**：重连成功率、异常恢复时间
4. **兼容性**：浏览器支持度、降级策略

## 二、实验一：音频采集性能对比

### 2.1 实验设计

**假设**：AudioWorklet + SharedArrayBuffer 方案在丢帧率和主线程影响上显著优于 ScriptProcessorNode 方案。

**变量控制**：
- 相同音频源（预录制音频文件）
- 相同采样率（16kHz）
- 相同处理逻辑（仅采集，无额外处理）
- 相同测试环境（Chrome 最新版，MacBook Pro）

### 2.2 实验代码

**传统方案**：

```javascript
// benchmark/script-processor.js
class ScriptProcessorBenchmark {
  constructor() {
    this.droppedFrames = 0;
    this.totalFrames = 0;
    this.processTimes = [];
    this.mainThreadBlocks = 0;
  }

  async run(duration = 60000) {
    const ctx = new AudioContext({ sampleRate: 16000 });
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const source = ctx.createMediaStreamSource(stream);
    const processor = ctx.createScriptProcessor(4096, 1, 1);

    const startTime = performance.now();

    processor.onaudioprocess = (e) => {
      const processStart = performance.now();

      // 模拟处理
      const input = e.inputBuffer.getChannelData(0);
      const output = new Int16Array(input.length);
      for (let i = 0; i < input.length; i++) {
        output[i] = Math.max(-32768, Math.min(32767, input[i] * 32768));
      }

      const processEnd = performance.now();
      const processTime = processEnd - processStart;
      this.processTimes.push(processTime);

      // 检测主线程阻塞
      if (processTime > 10) this.mainThreadBlocks++;

      this.totalFrames++;
    };

    source.connect(processor);
    processor.connect(ctx.destination);

    // 运行指定时长
    await new Promise(r => setTimeout(r, duration));

    // 清理
    processor.disconnect();
    source.disconnect();
    stream.getTracks().forEach(t => t.stop());
    await ctx.close();

    return this.getResults();
  }

  getResults() {
    const times = this.processTimes;
    times.sort((a, b) => a - b);

    return {
      totalFrames: this.totalFrames,
      droppedFrames: this.droppedFrames,
      dropRate: this.droppedFrames / this.totalFrames,
      mainThreadBlocks: this.mainThreadBlocks,
      blockRate: this.mainThreadBlocks / this.totalFrames,
      processTime: {
        p50: times[Math.floor(times.length * 0.5)],
        p90: times[Math.floor(times.length * 0.9)],
        p99: times[Math.floor(times.length * 0.99)],
        max: times[times.length - 1],
      },
    };
  }
}
```

```javascript
// benchmark/audio-worklet.js
class AudioWorkletBenchmark {
  constructor() {
    this.droppedFrames = 0;
    this.totalFrames = 0;
    this.processTimes = [];
  }

  async run(duration = 60000) {
    const ctx = new AudioContext({ sampleRate: 16000 });
    await ctx.audioWorklet.addModule('/capture-processor.js');

    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    const source = ctx.createMediaStreamSource(stream);
    const worklet = new AudioWorkletNode(ctx, 'vk-capture-processor', {
      processorOptions: {
        targetRate: 16000,
        capacity: 16000 * 4,
        enableSAB: true,
      },
    });

    // 接收处理结果
    worklet.port.onmessage = (e) => {
      if (e.data.type === 'stats') {
        this.totalFrames = e.data.framesEmitted;
        this.droppedFrames = e.data.droppedFrames;
        this.processTimes.push(e.data.processTime);
      }
    };

    source.connect(worklet);

    // 运行指定时长
    await new Promise(r => setTimeout(r, duration));

    // 清理
    worklet.disconnect();
    source.disconnect();
    stream.getTracks().forEach(t => t.stop());
    await ctx.close();

    return this.getResults();
  }

  getResults() {
    const times = this.processTimes;
    times.sort((a, b) => a - b);

    return {
      totalFrames: this.totalFrames,
      droppedFrames: this.droppedFrames,
      dropRate: this.droppedFrames / this.totalFrames,
      mainThreadBlocks: 0, // Worklet 不阻塞主线程
      blockRate: 0,
      processTime: {
        p50: times[Math.floor(times.length * 0.5)] || 0,
        p90: times[Math.floor(times.length * 0.9)] || 0,
        p99: times[Math.floor(times.length * 0.99)] || 0,
        max: times[times.length - 1] || 0,
      },
    };
  }
}
```

## 三、实验二：零拷贝性能对比

### 3.1 实验设计

**假设**：SharedArrayBuffer 零拷贝方案在 CPU 占用和吞吐量上优于 MessagePort 拷贝方案。

### 3.2 实验代码

```javascript
// benchmark/zero-copy.js
class ZeroCopyBenchmark {
  // MessagePort 拷贝方案
  async measureCopyTransfer(size, iterations) {
    const data = new Int16Array(size);
    for (let i = 0; i < size; i++) data[i] = Math.random() * 32767;

    const start = performance.now();
    let transferred = 0;

    for (let i = 0; i < iterations; i++) {
      // 模拟 MessagePort 传输（拷贝）
      const copy = new Int16Array(data);
      transferred += copy.byteLength;
    }

    const end = performance.now();
    const duration = end - start;

    return {
      duration,
      throughput: transferred / duration, // bytes/ms
      cpuTime: duration,
    };
  }

  // SharedArrayBuffer 零拷贝方案
  async measureZeroCopy(size, iterations) {
    const sab = new SharedArrayBuffer(size * 2);
    const data = new Int16Array(sab);
    for (let i = 0; i < size; i++) data[i] = Math.random() * 32767;

    const start = performance.now();
    let transferred = 0;

    for (let i = 0; i < iterations; i++) {
      // 零拷贝：直接访问
      const value = data[0]; // 模拟读取
      transferred += sab.byteLength;
    }

    const end = performance.now();
    const duration = end - start;

    return {
      duration,
      throughput: transferred / duration,
      cpuTime: duration,
    };
  }

  async run() {
    const sizes = [1024, 4096, 16384, 65536]; // 1KB, 4KB, 16KB, 64KB
    const iterations = 10000;

    console.log('=== 零拷贝性能对比实验 ===\n');

    const results = [];

    for (const size of sizes) {
      console.log(`测试数据大小: ${size} samples (${size * 2} bytes)`);

      const copyResult = await this.measureCopyTransfer(size, iterations);
      const zeroCopyResult = await this.measureZeroCopy(size, iterations);

      const improvement = ((copyResult.duration - zeroCopyResult.duration) / copyResult.duration * 100).toFixed(1);

      results.push({
        size: `${size * 2} bytes`,
        copyTime: copyResult.duration.toFixed(2),
        zeroCopyTime: zeroCopyResult.duration.toFixed(2),
        improvement: `${improvement}%`,
      });

      console.log(`  拷贝方案: ${copyResult.duration.toFixed(2)}ms`);
      console.log(`  零拷贝: ${zeroCopyResult.duration.toFixed(2)}ms`);
      console.log(`  提升: ${improvement}%\n`);
    }

    console.table(results);
    return results;
  }
}
```

## 四、实验三：重连策略对比

### 4.1 实验设计

**假设**：抖动指数退避策略在重连成功率和服务器压力上优于固定间隔策略。

### 4.2 模拟实验

```javascript
// benchmark/reconnect.js
class ReconnectBenchmark {
  // 固定间隔策略
  fixedInterval(attempt, baseMs) {
    return baseMs;
  }

  // 指数退避策略
  exponentialBackoff(attempt, baseMs, maxMs) {
    return Math.min(baseMs * Math.pow(2, attempt), maxMs);
  }

  // 抖动指数退避策略
  jitteredExponential(attempt, baseMs, maxMs) {
    const exp = baseMs * Math.pow(2, attempt);
    const jittered = exp * (0.5 + Math.random() * 0.5);
    return Math.min(jittered, maxMs);
  }

  // 模拟多客户端重连
  simulateClients(strategy, clientCount, serverCapacity) {
    const clients = [];
    let serverLoad = [];
    let successCount = 0;
    let totalWaitTime = 0;

    // 初始化客户端
    for (let i = 0; i < clientCount; i++) {
      clients.push({
        id: i,
        attempt: 0,
        nextRetry: 0,
        connected: false,
      });
    }

    // 模拟时间推进
    const maxTime = 60000; // 60秒
    for (let t = 0; t < maxTime; t += 100) {
      // 当前时刻尝试重连的客户端数
      const retrying = clients.filter(c => !c.connected && c.nextRetry <= t);

      // 服务器负载
      serverLoad.push(retrying.length);

      // 成功连接（不超过服务器容量）
      const canConnect = Math.min(retrying.length, serverCapacity);
      for (let i = 0; i < canConnect; i++) {
        retrying[i].connected = true;
        successCount++;
        totalWaitTime += t;
      }

      // 失败的客户端安排下次重连
      for (let i = canConnect; i < retrying.length; i++) {
        const client = retrying[i];
        client.attempt++;
        const delay = strategy(client.attempt, 100, 30000);
        client.nextRetry = t + delay;
      }
    }

    // 计算指标
    const peakLoad = Math.max(...serverLoad);
    const avgLoad = serverLoad.reduce((a, b) => a + b, 0) / serverLoad.length;
    const successRate = successCount / clientCount;
    const avgWaitTime = successCount > 0 ? totalWaitTime / successCount : 0;

    return {
      successRate,
      avgWaitTime,
      peakLoad,
      avgLoad,
    };
  }

  async run() {
    console.log('=== 重连策略对比实验 ===\n');

    const clientCount = 1000;
    const serverCapacity = 50; // 每时刻最多处理50个连接

    console.log(`模拟场景: ${clientCount}个客户端同时断开，服务器每时刻最多处理${serverCapacity}个连接\n`);

    // 固定间隔
    const fixedResult = this.simulateClients(
      this.fixedInterval,
      clientCount,
      serverCapacity
    );

    // 指数退避
    const expResult = this.simulateClients(
      this.exponentialBackoff,
      clientCount,
      serverCapacity
    );

    // 抖动指数退避
    const jitteredResult = this.simulateClients(
      this.jitteredExponential,
      clientCount,
      serverCapacity
    );

    console.table({
      '固定间隔': fixedResult,
      '指数退避': expResult,
      '抖动指数退避': jitteredResult,
    }, ['successRate', 'avgWaitTime', 'peakLoad', 'avgLoad']);

    console.log('\n关键发现:');
    console.log(`- 抖动指数退避的峰值负载最低: ${jitteredResult.peakLoad}`);
    console.log(`- 相比固定间隔降低: ${((fixedResult.peakLoad - jitteredResult.peakLoad) / fixedResult.peakLoad * 100).toFixed(1)}%`);

    return { fixedResult, expResult, jitteredResult };
  }
}
```

## 五、实验四：四路径去重效果验证

### 5.1 实验设计

**假设**：四路径去重算法能有效减少UI更新次数，提升渲染性能。

### 5.2 实验代码

```javascript
// benchmark/dedup.js
import { transcriptionReducer, initialTranscriptionState } from '@voice-kit/scene-transcribe';

class DedupBenchmark {
  // 无去重：直接更新
  noDedup(frames) {
    let updateCount = 0;
    let text = '';

    for (const frame of frames) {
      text = frame.text;
      updateCount++;
    }

    return { updateCount, finalText: text };
  }

  // 四路径去重
  withDedup(frames) {
    let state = initialTranscriptionState;
    let updateCount = 0;

    for (const frame of frames) {
      const action = {
        type: frame.isFinal ? 'FINAL' : 'PARTIAL',
        text: frame.text,
        ts: frame.ts,
        utterances: frame.utterances || [],
      };

      const nextState = transcriptionReducer(state, action);
      if (nextState !== state) {
        updateCount++;
      }
      state = nextState;
    }

    return { updateCount, finalText: state.cards.map(c => c.text).join('') };
  }

  // 生成模拟ASR帧
  generateFrames(scenario) {
    const frames = [];

    if (scenario === 'cumulative') {
      // 累积模式：每次发送完整文本
      const texts = ['你', '你好', '你好世', '你好世界'];
      texts.forEach((text, i) => {
        frames.push({ text, ts: i * 500, isFinal: false });
      });
    } else if (scenario === 'incremental') {
      // 增量模式：每次发送新增部分
      const texts = ['你', '好', '世', '界'];
      let accumulated = '';
      texts.forEach((text, i) => {
        accumulated += text;
        frames.push({ text: accumulated, ts: i * 500, isFinal: false });
      });
    } else if (scenario === 'mixed') {
      // 混合模式：部分+最终交替
      frames.push({ text: '今天天气', ts: 0, isFinal: false });
      frames.push({ text: '今天天气不错', ts: 500, isFinal: false });
      frames.push({ text: '今天天气不错', ts: 1000, isFinal: true, utterances: [{ text: '今天天气不错', startMs: 0, endMs: 1000 }] });
      frames.push({ text: '适合出门', ts: 1500, isFinal: false });
      frames.push({ text: '适合出门', ts: 2000, isFinal: true, utterances: [{ text: '适合出门', startMs: 1000, endMs: 2000 }] });
    }

    return frames;
  }

  async run() {
    console.log('=== 四路径去重效果验证 ===\n');

    const scenarios = ['cumulative', 'incremental', 'mixed'];

    for (const scenario of scenarios) {
      console.log(`\n场景: ${scenario}`);

      const frames = this.generateFrames(scenario);
      console.log(`帧数: ${frames.length}`);

      const noDedupResult = this.noDedup(frames);
      const withDedupResult = this.withDedup(frames);

      const reduction = ((noDedupResult.updateCount - withDedupResult.updateCount) / noDedupResult.updateCount * 100).toFixed(1);

      console.log(`  无去重更新次数: ${noDedupResult.updateCount}`);
      console.log(`  四路径去重更新次数: ${withDedupResult.updateCount}`);
      console.log(`  减少: ${reduction}%`);
    }

    // 大规模测试
    console.log('\n\n大规模测试 (1000帧累积模式):');
    const largeFrames = [];
    let text = '';
    for (let i = 0; i < 1000; i++) {
      text += '字';
      largeFrames.push({ text, ts: i * 100, isFinal: false });
    }

    const noDedupLarge = this.noDedup(largeFrames);
    const withDedupLarge = this.withDedup(largeFrames);

    console.log(`  无去重: ${noDedupLarge.updateCount} 次更新`);
    console.log(`  四路径去重: ${withDedupLarge.updateCount} 次更新`);
    console.log(`  减少: ${((noDedupLarge.updateCount - withDedupLarge.updateCount) / noDedupLarge.updateCount * 100).toFixed(1)}%`);
  }
}
```

## 六、实验五：端到端延迟测量

### 6.1 实验设计

**测量点**：

```
用户说话 ──①──> 麦克风采集 ──②──> 编码 ──③──> 网络发送
                                                    │
                                                    ↓
用户看到结果 <──⑦── 渲染 <──⑥── 解码 <──⑤── 网络接收 <──④── ASR处理
```

### 6.2 实验代码

```javascript
// benchmark/e2e-latency.js
import HdrHistogram from 'hdr-histogram-js';

class E2ELatencyBenchmark {
  constructor() {
    this.histogram = HdrHistogram.init();
    this.chunkSendTimes = new Map();
  }

  // 客户端：发送音频chunk
  sendAudioChunk(chunkId, audioData) {
    const sendTime = performance.now();
    this.chunkSendTimes.set(chunkId, sendTime);

    // 发送到服务器
    ws.send(JSON.stringify({
      type: 'audio',
      chunkId,
      data: Array.from(new Int16Array(audioData)),
    }));
  }

  // 客户端：接收ASR结果
  onResult(msg) {
    if (msg.type === 'result' && msg.chunkId) {
      const receiveTime = performance.now();
      const sendTime = this.chunkSendTimes.get(msg.chunkId);

      if (sendTime) {
        const latency = receiveTime - sendTime;
        this.histogram.recordValue(latency);
        this.chunkSendTimes.delete(msg.chunkId);
      }
    }
  }

  // 服务端模拟（实际测试时连接真实ASR服务）
  async mockServer() {
    // 模拟处理延迟：100-300ms
    const processingDelay = 100 + Math.random() * 200;
    await new Promise(r => setTimeout(r, processingDelay));
  }

  async run(duration = 60000) {
    console.log('=== 端到端延迟测量 ===\n');
    console.log(`运行时长: ${duration / 1000}秒\n`);

    const startTime = performance.now();
    let chunkId = 0;

    // 模拟音频采集和发送
    const sendInterval = setInterval(() => {
      // 每100ms发送一个chunk
      const audioData = new ArrayBuffer(3200); // 16kHz * 0.1s * 2bytes
      this.sendAudioChunk(chunkId++, audioData);
    }, 100);

    // 运行指定时长
    await new Promise(r => setTimeout(r, duration));

    clearInterval(sendInterval);

    // 输出结果
    console.log('延迟分布:');
    console.log(`  p50: ${this.histogram.getValueAtPercentile(50).toFixed(1)}ms`);
    console.log(`  p90: ${this.histogram.getValueAtPercentile(90).toFixed(1)}ms`);
    console.log(`  p99: ${this.histogram.getValueAtPercentile(99).toFixed(1)}ms`);
    console.log(`  max: ${this.histogram.getMaxValue().toFixed(1)}ms`);

    // 延迟分布直方图
    console.log('\n延迟分布直方图:');
    const percentiles = [10, 25, 50, 75, 90, 95, 99];
    const values = percentiles.map(p => this.histogram.getValueAtPercentile(p));

    const maxVal = Math.max(...values);
    const scale = 50 / maxVal;

    percentiles.forEach((p, i) => {
      const bar = '█'.repeat(Math.floor(values[i] * scale));
      console.log(`  p${p.toString().padStart(2)}: ${bar} ${values[i].toFixed(1)}ms`);
    });

    return {
      p50: this.histogram.getValueAtPercentile(50),
      p90: this.histogram.getValueAtPercentile(90),
      p99: this.histogram.getValueAtPercentile(99),
      max: this.histogram.getMaxValue(),
    };
  }
}
```

## 七、实验六：资源占用对比

### 7.1 实验设计

**测量指标**：

```javascript
// benchmark/resource-usage.js
class ResourceUsageBenchmark {
  async measureResourceUsage(scenario, duration = 60000) {
    // 使用 Performance API 测量
    const startMetrics = performance.measureMemory ? await performance.measureMemory() : null;

    const startTime = performance.now();
    const cpuSamples = [];

    // 定期采样CPU占用
    const cpuInterval = setInterval(() => {
      // 使用 Performance API 估算CPU占用
      const now = performance.now();
      const elapsed = now - startTime;
      const work = this.estimateWork();
      const cpuUsage = work / elapsed;
      cpuSamples.push(cpuUsage);
    }, 1000);

    // 运行场景
    await scenario.run(duration);

    clearInterval(cpuInterval);

    const endMetrics = performance.measureMemory ? await performance.measureMemory() : null;

    // 计算结果
    const avgCpu = cpuSamples.reduce((a, b) => a + b, 0) / cpuSamples.length;
    const peakCpu = Math.max(...cpuSamples);

    const memoryUsage = endMetrics && startMetrics
      ? (endMetrics.bytes - startMetrics.bytes) / 1024 / 1024
      : 0;

    return {
      avgCpu: (avgCpu * 100).toFixed(1) + '%',
      peakCpu: (peakCpu * 100).toFixed(1) + '%',
      memoryMB: memoryUsage.toFixed(1) + ' MB',
    };
  }

  estimateWork() {
    // 简化：使用 performance.now() 的调用次数估算
    return performance.now() % 1000;
  }

  async run() {
    console.log('=== 资源占用对比 ===\n');

    // 传统方案
    console.log('测量 ScriptProcessorNode 方案...');
    const scriptBenchmark = new ScriptProcessorBenchmark();
    const scriptResources = await this.measureResourceUsage(scriptBenchmark);

    await new Promise(r => setTimeout(r, 5000));

    // voice-kit 方案
    console.log('测量 AudioWorklet+SAB 方案...');
    const workletBenchmark = new AudioWorkletBenchmark();
    const workletResources = await this.measureResourceUsage(workletBenchmark);

    console.log('\n资源占用对比:');
    console.table({
      'ScriptProcessor': scriptResources,
      'AudioWorklet+SAB': workletResources,
    });

    return { scriptResources, workletResources };
  }
}
```
