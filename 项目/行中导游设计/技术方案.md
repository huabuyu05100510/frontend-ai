# 注意：只能正常工作在python 3.12环境下，如果是3.13环境则会导致请求失败
import argparse
import datetime
import json
import os
import sys
from concurrent.futures import ThreadPoolExecutor


import requests
# 引入 sseclient-py 库
import sseclient
BASE_URL = "http://10.191.156.8:8065/"


def request(data):
    """
    向流式 API (/api/chat/stream) 发送 GET 请求并处理响应。
    """
    # 准备 GET 请求的查询参数
    params = {"prompt": data}
    full_response_content = []


    try:
        # 使用 GET 请求，并设置 stream=True 来接收流式响应
        response = requests.get(
            f"{BASE_URL}/api/chat/stream",
            params=params,
            stream=True,
            timeout=600
        )
        # 检查请求是否成功 (例如，状态码 200)
        response.raise_for_status()


        # 使用 sseclient 来解析事件流
        client = sseclient.SSEClient(response)


        # 实时打印流式输出，给用户即时反馈
        print(f"\n--- Streaming response for: {data[:50]}... ---")
        for event in client.events():
            # event: complete 表示流结束
            if event.event == 'complete':
                # print("\nStream completed.") # 可以取消注释来查看结束信息
                break
            # event: error 表示服务端出错
            elif event.event == 'error':
                error_info = json.loads(event.data)
                print(f"\nServer-side error: {error_info.get('error')}")
                return None  # 返回 None 表示失败
            # 默认事件是 'message'，我们处理其 data 字段
            elif event.data:
                try:
                    # 服务端发送的是 JSON 字符串，所以需要解析
                    chunk_data = json.loads(event.data)


                    # 根据 chunk 的实际结构提取文本内容
                    # 这是一个健壮的设计，尝试从常见的键（'content', 'text', 'token'）中提取
                    # 如果 chunk 本身就是字符串，则直接使用
                    text_part = ""
                    if isinstance(chunk_data, dict):
                        text_part = chunk_data.get("content", chunk_data.get("text", chunk_data.get("token", "")))
                    elif isinstance(chunk_data, str):
                        text_part = chunk_data


                    if text_part:
                        # 实时打印到控制台
                        sys.stdout.write(text_part)
                        sys.stdout.flush()
                        # 收集所有文本片段
                        full_response_content.append(text_part)


                except json.JSONDecodeError:
                    # 如果数据不是有效的 JSON，打印警告并跳过
                    print(f"\nWarning: Could not decode JSON from data: {event.data}")
                    continue


        print("\n--- End of stream ---")
        # 将所有文本片段合并成一个完整的响应
        return "".join(full_response_content)


    except requests.exceptions.Timeout:
        print(f"\n请求超时！请稍后再试: {data[:30]}...")
        return None
    except requests.exceptions.RequestException as e:
        print(f"\n请求失败：{e}")
        return None
    except Exception as e:
        print(f"\n处理流时发生未知错误: {e}")
        return None


if __name__ == "__main__":
    request("去北京西站要多久?")
